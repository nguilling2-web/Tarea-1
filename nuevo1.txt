class RegistroTemperaturas:

    def __init__(self):
        self.temperaturas = []

    def agregar(self, *temperaturas):
        for temperatura in temperaturas:
            self.temperaturas.append(temperatura)

    def temperatura_maxima(self):
        mayor = self.temperaturas[0]

        for temperatura in self.temperaturas:
            if temperatura > mayor:
                mayor = temperatura

        return mayor

    def temperatura_minima(self):
        menor = self.temperaturas[0]

        for temperatura in self.temperaturas:
            if temperatura < menor:
                menor = temperatura

        return menor

    def promedio(self):
        suma = 0

        for temperatura in self.temperaturas:
            suma = suma + temperatura

        return suma / len(self.temperaturas)

"""Clase ContadorCaracteres que:

Tenga método analizar(texto) que guarde cada carácter en un diccionario contando repeticiones.
Tenga método caracter_mas_repetido() que retorne el carácter con mayor frecuencia.
Tenga método frecuencia(caracter) que retorne cuántas veces aparece."""



class ContadorCaracteres:

    def __init__(self):
        self.caracteres = {}

    def analizar(self, texto):

        for caracter in texto:
            if caracter in self.caracteres:
                self.caracteres[caracter] = self.caracteres[caracter] + 1
            else:
                self.caracteres[caracter] = 1

    def caracter_mas_repetido(self):
        caracter_mayor = ""
        cantidad_mayor = 0

        for caracter, cantidad in self.caracteres.items():
            if cantidad > cantidad_mayor:
                cantidad_mayor = cantidad
                caracter_mayor = caracter

        return caracter_mayor

    def frecuencia(self, caracter):
        if caracter in self.caracteres:
            return self.caracteres[caracter]

        return 0

print(cc.caracteres)

"""Clase CombinadorDatos que:

Tenga método combinar(lista1, lista2) que retorne una lista alternando los elementos.
Tenga método combinar_multiples(*listas) que combine varias listas.
Tenga método sin_repetidos(lista) que retorne una lista sin duplicados usando un conjunto."""


class CombinadorDatos:

    def combinar(self, lista1, lista2):
        resultado = []

        mayor = max(len(lista1), len(lista2))

        for i in range(mayor):

            if i < len(lista1):
                resultado.append(lista1[i])

            if i < len(lista2):
                resultado.append(lista2[i])

        return resultado

    def combinar_multiples(self, *listas):
        resultado = []

        mayor = 0

        for lista in listas:
            if len(lista) > mayor:
                mayor = len(lista)

        for i in range(mayor):

            for lista in listas:
                if i < len(lista):
                    resultado.append(lista[i])

        return resultado

    def sin_repetidos(self, lista):
        conjunto = set(lista)

        return list(conjunto)


cd = CombinadorDatos()

print(cd.combinar([1, 2, 3], ["a", "b", "c"]))

print(cd.combinar_multiples(
    [1, 2, 3],
    ["a", "b", "c"],
    ["x", "y", "z"]
))

print(cd.sin_repetidos([1, 2, 2, 3, 3, 4]))
